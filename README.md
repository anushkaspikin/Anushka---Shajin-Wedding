<script>
(() => {
  // Countdown is pinned to the wedding ceremony time in India (UTC+05:30).
  const weddingTime = new Date('2027-01-18T10:30:00+05:30').getTime();
  const pad = n => String(n).padStart(2,'0');
  function updateCountdown(){
    let diff = Math.max(0, weddingTime - Date.now());
    const d = Math.floor(diff / 86400000); diff %= 86400000;
    const h = Math.floor(diff / 3600000); diff %= 3600000;
    const m = Math.floor(diff / 60000); diff %= 60000;
    const s = Math.floor(diff / 1000);
    document.getElementById('days').textContent = d;
    document.getElementById('hours').textContent = pad(h);
    document.getElementById('minutes').textContent = pad(m);
    document.getElementById('seconds').textContent = pad(s);
  }
  updateCountdown(); window.setInterval(updateCountdown,1000);

  // Automatic scroll is gentle, starts after a pause, and stops on any visitor interaction.
  const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  let autoScrollTimer = null, autoScrollFrame = null, stopped = false, lastFrame = 0;
  function stopAutoScroll(){
    if(stopped) return;
    stopped = true;
    if(autoScrollTimer) clearTimeout(autoScrollTimer);
    if(autoScrollFrame) cancelAnimationFrame(autoScrollFrame);
    ['pointerdown','touchstart','touchmove','wheel','keydown','click'].forEach(type =>
      window.removeEventListener(type, stopAutoScroll, {capture:true})
    );
  }
  ['pointerdown','touchstart','touchmove','wheel','keydown','click'].forEach(type =>
    window.addEventListener(type, stopAutoScroll, {capture:true, passive:true})
  );
  function tick(timestamp){
    if(stopped) return;
    if(!lastFrame) lastFrame = timestamp;
    const elapsed = Math.min(timestamp - lastFrame, 50);
    lastFrame = timestamp;
    window.scrollBy(0, elapsed * 0.014); // about 14 px/sec
    if(window.scrollY + window.innerHeight >= document.documentElement.scrollHeight - 2){
      stopAutoScroll(); return;
    }
    autoScrollFrame = requestAnimationFrame(tick);
  }
  if(!reduceMotion){
    autoScrollTimer = window.setTimeout(() => {
      if(!stopped) autoScrollFrame = requestAnimationFrame(tick);
    }, 3200);
  }

  const music = document.getElementById('weddingMusic');
  const musicButton = document.getElementById('musicButton');
  const musicStatus = document.getElementById('musicStatus');
  musicButton.addEventListener('click', async () => {
    if(music.paused){
      try{
        await music.play();
        musicButton.textContent = 'Ⅱ Pause music';
        musicStatus.textContent = 'Music is playing.';
      }catch(e){
        musicStatus.textContent = 'Music is not added yet. Upload a licensed MP3 as assets/song.mp3 in the repository, then redeploy.';
      }
    }else{
      music.pause(); musicButton.textContent = '♫ Play music'; musicStatus.textContent = 'Music paused.';
    }
  });
  document.getElementById('shareButton').addEventListener('click', async () => {
    const shareData = {title:document.title, text:'You are warmly invited to the wedding of Anushka & Shajin Jose on 18 January 2027.', url:window.location.href};
    try{
      if(navigator.share) await navigator.share(shareData);
      else if(navigator.clipboard && window.isSecureContext){
        await navigator.clipboard.writeText(window.location.href);
        document.getElementById('shareButton').textContent = 'Link copied ✓';
      }else{
        window.prompt('Copy your wedding invitation link:', window.location.href);
      }
    }catch(e){ /* visitor cancelled sharing */ }
  });
})();
